2026跨年烟花 | 新年快乐

2026跨年快乐 🎆

2026新年快乐 🎉 万事顺意 · 平安喜乐 
<script>
    // ====================== 倒计时功能（仅显示剩余时间，无结束文字变化） ======================
    // 目标时间：2026年1月1日 00:00:00
    const targetTime = new Date('2026-01-01 00:00:00').getTime();
    const countdownEl = document.getElementById('countdown');

    // 更新倒计时（无论是否结束，都显示时分秒，标题不变）
    function updateCountdown() {
        const now = new Date().getTime();
        const diff = targetTime - now;

        // 计算时/分/秒（倒计时结束后显示00:00:00）
        let hours = 0, minutes = 0, seconds = 0;
        if (diff > 0) {
            hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
            seconds = Math.floor((diff % (1000 * 60)) / 1000);
        }

        // 补零（比如 9 → 09）
        const formatNum = (num) => num.toString().padStart(2, '0');
        countdownEl.textContent = `${formatNum(hours)}:${formatNum(minutes)}:${formatNum(seconds)}`;
    }
    // 初始化+每秒更新
    updateCountdown();
    setInterval(updateCountdown, 1000);

    // ====================== 雪花特效 ======================
    const snowCanvas = document.getElementById('snow-canvas');
    const snowCtx = snowCanvas.getContext('2d');
    let snowflakes = [];

    // 适配雪花画布大小
    function resizeSnowCanvas() {
        snowCanvas.width = window.innerWidth;
        snowCanvas.height = window.innerHeight;
    }
    resizeSnowCanvas();
    window.addEventListener('resize', resizeSnowCanvas);

    // 雪花类
    class Snowflake {
        constructor() {
            this.x = Math.random() * snowCanvas.width;
            this.y = Math.random() * snowCanvas.height;
            this.size = Math.random() * 3 + 1; // 大小1-4px
            this.speed = Math.random() * 2 + 0.5; // 下落速度0.5-2.5px
            this.wind = Math.random() * 1 - 0.5; // 左右飘动-0.5~0.5px
            this.alpha = Math.random() * 0.5 + 0.5; // 透明度0.5-1
        }
        update() {
            this.y += this.speed;
            this.x += this.wind;
            // 雪花落到底部，重新从顶部生成
            if (this.y > snowCanvas.height) {
                this.y = -this.size;
                this.x = Math.random() * snowCanvas.width;
            }
        }
        draw() {
            snowCtx.save();
            snowCtx.globalAlpha = this.alpha;
            snowCtx.beginPath();
            snowCtx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
            snowCtx.fillStyle = '#fff';
            snowCtx.fill();
            snowCtx.restore();
        }
    }

    // 生成雪花（数量根据屏幕宽度调整）
    function createSnowflakes() {
        const count = Math.floor(snowCanvas.width / 5); // 每5px一个雪花
        snowflakes = [];
        for (let i = 0; i < count; i++) {
            snowflakes.push(new Snowflake());
        }
    }
    createSnowflakes();

    // 雪花动画循环
    function animateSnow() {
        snowCtx.clearRect(0, 0, snowCanvas.width, snowCanvas.height);
        snowflakes.forEach(flake => {
            flake.update();
            flake.draw();
        });
        requestAnimationFrame(animateSnow);
    }
    animateSnow();

    // ====================== 升级烟花特效 ======================
    const fireCanvas = document.getElementById('fireworks-canvas');
    const fireCtx = fireCanvas.getContext('2d');

    // 适配烟花画布大小
    function resizeFireCanvas() {
        fireCanvas.width = window.innerWidth;
        fireCanvas.height = window.innerHeight;
    }
    resizeFireCanvas();
    window.addEventListener('resize', resizeFireCanvas);

    // 工具函数
    const random = (min, max) => Math.random() * (max - min) + min;
    const floorRandom = (min, max) => Math.floor(random(min, max));

    // 升级配色（跨年喜庆色：红/金/粉/蓝/紫）
    const getRandomColor = () => {
        const colors = [
            '#ff0040', '#ff7f00', '#ffd700', '#ffff00', // 红/橙/金/黄
            '#00ff00', '#00ffff', '#0000ff', '#8b00ff'  // 绿/青/蓝/紫
        ];
        return colors[floorRandom(0, colors.length)];
    };

    // 烟花轨迹类（新增：烟花升空轨迹）
    class FireworkTrail {
        constructor(x, y) {
            this.x = x;
            this.y = y;
            this.targetY = random(50, fireCanvas.height / 2); // 升空目标高度
            this.speed = random(8, 15); // 升空速度
            this.color = getRandomColor();
            this.alpha = 1;
        }
        update() {
            this.y -= this.speed; // 向上飞
            this.alpha -= 0.005; // 渐隐
            // 到达目标高度，爆炸生成粒子
            if (this.y <= this.targetY) {
                createFireworkParticles(this.x, this.y);
                return false; // 移除轨迹
            }
            return true;
        }
        draw() {
            fireCtx.save();
            fireCtx.globalAlpha = this.alpha;
            fireCtx.beginPath();
            fireCtx.moveTo(this.x, fireCanvas.height); // 从底部开始
            fireCtx.lineTo(this.x, this.y); // 到当前位置
            fireCtx.strokeStyle = this.color;
            fireCtx.lineWidth = 2;
            fireCtx.stroke();
            fireCtx.restore();
        }
    }

    // 烟花粒子类（升级：更大、更亮、有拖影）
    class FireworkParticle {
        constructor(x, y) {
            this.x = x;
            this.y = y;
            this.speed = random(2, 8); // 粒子速度更快
            this.angle = random(0, Math.PI * 2);
            this.gravity = 0.08; // 重力更强，下落更自然
            this.vx = Math.cos(this.angle) * this.speed;
            this.vy = Math.sin(this.angle) * this.speed;
            this.color = getRandomColor();
            this.alpha = 1;
            this.decay = random(0.008, 0.015); // 衰减更快，烟花更短暂
            this.size = random(2, 4); // 粒子更大
        }
        update() {
            this.vy += this.gravity;
            this.x += this.vx;
            this.y += this.vy;
            this.alpha -= this.decay;
            // 粒子减速
            this.vx *= 0.98;
            this.vy *= 0.98;
        }
        draw() {
            fireCtx.save();
            fireCtx.globalAlpha = this.alpha;
            fireCtx.beginPath();
            // 圆形粒子+轻微模糊（拖影效果）
            fireCtx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
            fireCtx.fillStyle = this.color;
            fireCtx.fill();
            fireCtx.restore();
        }
    }

    let trails = []; // 烟花轨迹数组
    let particles = []; // 粒子数组

    // 生成烟花粒子
    function createFireworkParticles(x, y) {
        const particleCount = floorRandom(150, 300); // 粒子更多，烟花更密
        for (let i = 0; i < particleCount; i++) {
            particles.push(new FireworkParticle(x, y));
        }
    }

    // 生成烟花轨迹（点击/自动）
    function createFireworkTrail(x, y) {
        trails.push(new FireworkTrail(x, y));
    }

    // 自动生成烟花（升级：随机位置+更频繁）
    setInterval(() => {
        const x = random(50, fireCanvas.width - 50);
        createFireworkTrail(x, fireCanvas.height);
    }, 800); // 每0.8秒一个

    // 点击生成烟花
    fireCanvas.addEventListener('click', (e) => {
        createFireworkTrail(e.clientX, fireCanvas.height);
    });

    // 烟花动画循环
    function animateFireworks() {
        // 半透明背景，保留烟花拖影
        fireCtx.fillStyle = 'rgba(0, 0, 0, 0.05)';
        fireCtx.fillRect(0, 0, fireCanvas.width, fireCanvas.height);
        
        // 更新并绘制轨迹
        trails = trails.filter(trail => {
            const alive = trail.update();
            if (alive) trail.draw();
            return alive;
        });

        // 更新并绘制粒子
        particles = particles.filter(particle => {
            particle.update();
            particle.draw();
            return particle.alpha > 0;
        });

        requestAnimationFrame(animateFireworks);
    }
    animateFireworks();
</script>
